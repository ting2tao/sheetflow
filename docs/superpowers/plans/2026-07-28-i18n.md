# SheetFlow Chinese/English Internationalization Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add complete Simplified Chinese and English UI, API-message, and SEO support at `/zh/` and `/en/` without losing in-progress application state when switching languages.

**Architecture:** `vue-i18n` owns frontend copy and interpolation, while a small locale module owns URL detection, preference persistence, and History API updates. FastAPI persists stable message codes plus parameters instead of rendered Chinese sentences. Vite builds neutral, Chinese, and English HTML entries that share one Vue bundle, and Nginx routes each locale subtree to the matching entry.

**Tech Stack:** Vue 3, vue-i18n 11, Vite 5, Vitest 2, Vue Test Utils 2, FastAPI, pytest, Nginx

Create `frontend/src/i18n/locale.js`:

```js
export const ZH_LOCALE = 'zh-CN'
export const EN_LOCALE = 'en-US'
export const DEFAULT_LOCALE = ZH_LOCALE
export const SUPPORTED_LOCALES = [ZH_LOCALE, EN_LOCALE]
export const LOCALE_STORAGE_KEY = 'sheetflow.locale'

const pathSegmentForLocale = {
  [ZH_LOCALE]: 'zh',
  [EN_LOCALE]: 'en',
}

export function localeFromPath(pathname) {
  const segment = pathname.split('/').filter(Boolean)[0]
  if (segment === 'zh') return ZH_LOCALE
  if (segment === 'en') return EN_LOCALE
  return null
}

function normalizeSavedLocale(locale) {
  return SUPPORTED_LOCALES.includes(locale) ? locale : null
}

function readSavedLocale() {
  try {
    return localStorage.getItem(LOCALE_STORAGE_KEY)
  } catch {
    return null
  }
}

function saveLocale(locale) {
  try {
    localStorage.setItem(LOCALE_STORAGE_KEY, locale)
  } catch {
    // The URL remains the source of truth when storage is unavailable.
  }
}

function localeFromBrowser(languages) {
  if (!languages.length) return DEFAULT_LOCALE
  return languages.some(language => language.toLowerCase().startsWith('zh'))
    ? ZH_LOCALE
    : EN_LOCALE
}

export function resolveInitialLocale({
  pathname = window.location.pathname,
  savedLocale = readSavedLocale(),
  browserLanguages = navigator.languages || [navigator.language].filter(Boolean),
} = {}) {
  return localeFromPath(pathname)
    || normalizeSavedLocale(savedLocale)
    || localeFromBrowser(browserLanguages)
}

export function localizedPath(locale, pathname = window.location.pathname) {
  const segments = pathname.split('/').filter(Boolean)
  if (segments[0] === 'zh' || segments[0] === 'en') segments.shift()
  const suffix = segments.length ? `${segments.join('/')}/` : ''
  return `/${pathSegmentForLocale[locale]}/${suffix}`
}

export function switchLocale(locale) {
  if (!SUPPORTED_LOCALES.includes(locale)) return
  saveLocale(locale)
  history.pushState(history.state, '', localizedPath(locale))
  window.dispatchEvent(new CustomEvent('sheetflow:locale-change', { detail: locale }))
}
```

```js
function hasTranslation(t, key) {
  return t(key) !== key
}

export function localizeJobMessage(payload, t) {
  const code = payload?.message_code
  const key = code && `job.${code.replace(/^job\./, '')}`
  if (key && hasTranslation(t, key)) {
    const count = payload.message_params?.sheets
    return count === undefined
      ? t(key, payload.message_params || {})
      : t(key, payload.message_params || {}, count)
  }
  if (payload?.message) return payload.message
  return t('errors.generic')
}

export function localizeApiError(payload, t) {
  const detail = payload?.detail
  if (detail && typeof detail === 'object' && detail.code) {
    const key = `errors.${detail.code}`
    if (hasTranslation(t, key)) return t(key, detail.params || {})
  }
  if (typeof detail === 'string') return detail
  return t('errors.generic')
}
```
